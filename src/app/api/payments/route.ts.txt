// src/app/api/payments/route.ts
import { NextRequest, NextResponse } from 'next/server';
import Stripe from 'stripe';
import { createClient } from '@supabase/supabase-js';

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, { apiVersion: '2022-11-15' });
const supabase = createClient(process.env.SUPABASE_URL!, process.env.SUPABASE_SERVICE_KEY!);

export async function POST(req: NextRequest) {
  const { studentId, courseId, amount, method } = await req.json();

  let paymentUrl = null;

  if (method === 'stripe') {
    const session = await stripe.checkout.sessions.create({
      payment_method_types: ['card'],
      line_items: [{
        price_data: {
          currency: 'usd',
          product_data: { name: `Course: ${courseId}` },
          unit_amount: amount,
        },
        quantity: 1,
      }],
      mode: 'payment',
      success_url: `${process.env.FRONTEND_URL}/payments/success?session_id={CHECKOUT_SESSION_ID}`,
      cancel_url: `${process.env.FRONTEND_URL}/payments/cancel`,
    });
    paymentUrl = session.url;
  } else if (method === 'paypal') {
    // TODO: integrate PayPal SDK here and generate approval URL
    paymentUrl = 'https://paypal.com/checkout/...';
  } else if (method === 'mobilemoney') {
    // TODO: integrate Mobile Money API (MTN/Orange) to generate payment link or USSD code
    paymentUrl = 'https://mobilemoney.com/payment/...';
  } else if (method === 'cash') {
    // For cash, no external URL; payment is confirmed manually
    paymentUrl = null;
  }

  // Save payment record
  const { data, error } = await supabase.from('payments').insert({
    student_id: studentId,
    course_id: courseId,
    amount,
    method,
    status: method === 'cash' ? 'pending_cash' : 'pending',
    external_url: paymentUrl,
  }).select('*');

  return NextResponse.json({ paymentUrl, record: data?.[0] });
}
